% Logo de parceiro, usado na folha de rosto e na contracapa.
% \logoparceiro[<altura>]{<arquivo sem extensão>}{<nome do parceiro>}
%   caixa de largura fixa (grade); sem o arquivo, sai o nome em texto.
% \logoparceiro*[<altura>]{<arquivo>}{<nome>}
%   largura natural (linha de logos); sem o arquivo, não sai nada.
% Procura Logos/<nome>.pdf, .png, .jpg ou .eps.
\makeatletter
\newcommand{\logo@arquivo}[2]{% #1 = arquivo sem extensão, #2 = opções
    \IfFileExists{#1.pdf}{\includegraphics[#2]{#1.pdf}}{%
    \IfFileExists{#1.png}{\includegraphics[#2]{#1.png}}{%
    \IfFileExists{#1.jpg}{\includegraphics[#2]{#1.jpg}}{%
    \IfFileExists{#1.eps}{\includegraphics[#2]{#1.eps}}{\relax}}}}%
}
\newcommand{\logo@existe}[3]{% #1 = arquivo, #2 = se existe, #3 = se não
    \IfFileExists{#1.pdf}{#2}{\IfFileExists{#1.png}{#2}{%
    \IfFileExists{#1.jpg}{#2}{\IfFileExists{#1.eps}{#2}{#3}}}}%
}
\NewDocumentCommand{\logoparceiro}{s O{2cm} m m}{%
    \IfBooleanTF{#1}{%
        \logo@existe{#3}{\hspace{0.2cm}\logo@arquivo{#3}{height=#2,width=2.5\dimexpr#2\relax,keepaspectratio}\hspace{0.2cm}}{}%
    }{%
        \makebox[2.6\dimexpr#2\relax][c]{%
            \logo@existe{#3}%
                {\logo@arquivo{#3}{height=#2,width=2.5\dimexpr#2\relax,keepaspectratio}}%
                {\parbox[c][#2][c]{2.5\dimexpr#2\relax}{\centering\bfseries\footnotesize #4}}%
        }%
    }%
}
\makeatother
